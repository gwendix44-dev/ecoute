// Serveur privé « Écoute » — Cloudflare Worker
// Routes (toutes protégées par l'en-tête X-App-Key) :
//   GET  /ping      test de connexion
//   POST /plan      plan du cours (titre, parties)
//   POST /section   écrit UNE partie du cours
//   POST /script    cours complet (ancienne route, compatibilité)
//   POST /tts       voix (Gemini TTS, 1 ou 2 voix)
// Secrets : GEMINI_API_KEY, APP_KEY. Binding : AI (texte de secours).

const cors = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'Content-Type, X-App-Key',
  'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
  'Access-Control-Expose-Headers': 'X-Engine',
};
const json = (obj, status = 200) => new Response(JSON.stringify(obj), { status, headers: { 'Content-Type': 'application/json', ...cors } });
const sleep = (ms) => new Promise((res) => setTimeout(res, ms));

export default {
  async fetch(req, env) {
    if (req.method === 'OPTIONS') return new Response(null, { headers: cors });
    const url = new URL(req.url);
    if (!env.APP_KEY || req.headers.get('X-App-Key') !== env.APP_KEY) return json({ error: "Code d'accès invalide" }, 401);
    try {
      if (url.pathname === '/ping') return json({ ok: true });
      if (req.method === 'POST') {
        if (url.pathname === '/plan') return await handlePlan(req, env);
        if (url.pathname === '/section') return await handleSection(req, env);
        if (url.pathname === '/script') return await handleScript(req, env);
        if (url.pathname === '/tts') return await handleTTS(req, env);
      }
      return json({ error: 'Not found' }, 404);
    } catch (e) {
      return json({ error: String((e && e.message) || e) }, 500);
    }
  },
};

const SYSTEM = `Tu es un auteur de cours audio en français, pour une seule auditrice/auditeur qui écoute dans le bus et ne doit jamais décrocher.
Règles absolues :
- Texte fait pour être DIT à voix haute : phrases courtes, rythme vivant, vocabulaire simple, aucune liste, aucun markdown, aucun symbole bizarre. Écris les sigles comme on les prononce (ex: « S and P cinq cents », « I A », « euros »). Écris les nombres en chiffres seulement s'ils sont courts, sinon en toutes lettres.
- Ton : un mélange de pote calé qui t'explique tranquillement (cool, un peu d'humour léger, jamais lourd) et de documentaire sérieux et précis. Jamais robotique, jamais scolaire.
- Explique de A à Z : une accroche forte dans les 15 premières secondes, puis le pourquoi, puis le comment, étape par étape, avec une analogie concrète par idée difficile. Chaque mot technique est expliqué la première fois qu'il apparaît.
- Relance l'attention toutes les ~2 minutes : question rhétorique, mini-récap (« on a vu trois choses »), petit rebondissement ou chiffre qui surprend.
- Les faits doivent être exacts. Si un chiffre est incertain, dis « environ » ou évite-le. N'invente rien.
Formats (choisis le meilleur selon le sujet) :
- « prof » : une seule voix (speaker A) qui explique. Pour les mécanismes techniques et les process.
- « duo » : deux voix A et B qui discutent comme dans un podcast : B pose les questions naïves que l'auditeur se pose, A explique, B reformule, ils se relancent. Pour les concepts à démystifier (finance, économie, idées reçues).
- « story » : une seule voix (speaker A) qui raconte comme un documentaire, avec une histoire, des personnages, de la tension. Pour l'histoire, les événements, les personnes, les inventions.
Réponds uniquement par le JSON demandé.`;

const OUTLINE_SCHEMA = {
  type: 'OBJECT',
  properties: {
    title: { type: 'STRING' },
    emoji: { type: 'STRING' },
    format: { type: 'STRING', enum: ['prof', 'duo', 'story'] },
    summary: { type: 'STRING' },
    next: { type: 'ARRAY', items: { type: 'STRING' } },
    sections: { type: 'ARRAY', items: { type: 'OBJECT', properties: { heading: { type: 'STRING' }, points: { type: 'STRING' } }, required: ['heading', 'points'] } },
  },
  required: ['title', 'emoji', 'format', 'summary', 'next', 'sections'],
};

const SEG_SCHEMA = {
  type: 'OBJECT',
  properties: {
    segments: {
      type: 'ARRAY',
      items: {
        type: 'OBJECT',
        properties: { speaker: { type: 'STRING', enum: ['A', 'B'] }, text: { type: 'STRING' }, pause: { type: 'NUMBER' } },
        required: ['speaker', 'text'],
      },
    },
  },
  required: ['segments'],
};

const LEVELS = {
  debutant: 'Débutant total : zéro jargon non expliqué, beaucoup d\'analogies du quotidien.',
  inter: 'Intermédiaire : l\'auditeur connaît les bases, va plus vite sur l\'essentiel et approfondit les mécanismes.',
  avance: 'Avancé : va dans le détail technique et les nuances, sans reexpliquer les bases.',
};

// 1) Gemini (modèles rapides, un seul essai par modèle sur quota), 2) IA Cloudflare en secours.

const GEM_DEAD = new Map();
const lower = (s) => {
  if (Array.isArray(s)) return s.map(lower);
  if (s && typeof s === 'object') { const o = {}; for (const k in s) o[k] = k === 'type' && typeof s[k] === 'string' ? s[k].toLowerCase() : lower(s[k]); return o; }
  return s;
};
const skeleton = (s) => {
  if (!s) return 'texte';
  const t = String(s.type || '').toLowerCase();
  if (t === 'object') { const o = {}; for (const k in s.properties || {}) o[k] = skeleton(s.properties[k]); return o; }
  if (t === 'array') return [skeleton(s.items), skeleton(s.items)];
  if (s.enum) return s.enum.join(' | ');
  if (t === 'integer' || t === 'number') return 0;
  return 'texte';
};

function makeLLM(env) {
  const models = [env.GEMINI_MODEL, 'gemini-3.8-flash', 'gemini-3.7-flash', 'gemini-3.1-flash-lite', 'gemini-flash-lite-latest', 'gemini-flash-latest'].filter((m, i, a) => m && a.indexOf(m) === i);
  const errs = [];
  const gemini = async (model, prompt, schema) => {
    const send = (thinking) => {
      const generationConfig = { responseMimeType: 'application/json', responseSchema: schema, temperature: 0.9, maxOutputTokens: 8000 };
      if (thinking) generationConfig.thinkingConfig = thinking;
      return fetch(`https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'x-goog-api-key': env.GEMINI_API_KEY },
        signal: AbortSignal.timeout(45000),
        body: JSON.stringify({ systemInstruction: { parts: [{ text: SYSTEM }] }, contents: [{ role: 'user', parts: [{ text: prompt }] }], generationConfig }),
      });
    };
    for (let a = 0; a < 2; a++) {
      let r;
      try {
        r = await send({ thinkingLevel: 'minimal' });
        if (r.status === 400) r = await send(null);
      } catch (e) { errs.push(`${model}: délai dépassé`); return null; }
      if (r.ok) {
        const d = await r.json();
        const raw = (d && d.candidates && d.candidates[0] && d.candidates[0].content && d.candidates[0].content.parts || []).map((p) => p.text || '').join('');
        try { return JSON.parse(raw); } catch { errs.push(`${model}: réponse illisible`); continue; }
      }
      const body = (await r.text()).slice(0, 300);
      errs.push(`${model}: ${r.status}`);
      if (r.status === 429) { GEM_DEAD.set(model, Date.now() + (/PerDay/i.test(body) ? 3600000 : 60000)); return null; }
      if ([500, 502, 503, 504, 524].includes(r.status)) { await sleep(600); continue; }
      return null;
    }
    return null;
  };
  const workersAI = async (prompt, schema) => {
    if (!env.AI) return null;
    const msgs = [{ role: 'system', content: SYSTEM }, { role: 'user', content: prompt }];
    const attempts = [
      { messages: msgs, response_format: { type: 'json_schema', json_schema: lower(schema) }, max_tokens: 6000 },
      { messages: [msgs[0], { role: 'user', content: prompt + '\n\nRéponds UNIQUEMENT par un objet JSON de la même forme que cet exemple (remplace les valeurs, garde les clés) : ' + JSON.stringify(skeleton(schema)) }], max_tokens: 6000 },
    ];
    for (const body of attempts) {
      try {
        const rr = await env.AI.run('@cf/meta/llama-3.3-70b-instruct-fp8-fast', body);
        let out = rr && rr.response;
        if (out && typeof out === 'object') return out;
        out = String(out || '');
        const i = out.indexOf('{'), j = out.lastIndexOf('}');
        if (i >= 0 && j > i) return JSON.parse(out.slice(i, j + 1));
      } catch (e) { errs.push('IA secours: ' + String((e && e.message) || e).slice(0, 120)); }
    }
    return null;
  };
  return async (prompt, schema, check) => {
    for (const m of models) {
      if ((GEM_DEAD.get(m) || 0) > Date.now()) continue;
      const out = await gemini(m, prompt, schema);
      if (out && (!check || check(out))) return out;
    }
    const out = await workersAI(prompt, schema);
    if (out && (!check || check(out))) return out;
    throw new Error('Service de texte indisponible (' + errs.slice(-3).join(' ; ') + ')');
  };
}

const ctxOf = (b) => {
  const level = LEVELS[b.level] ? b.level : 'debutant';
  let ctx = `Sujet : ${String(b.topic).slice(0, 300)}\nNiveau : ${LEVELS[level]}\n` + (!b.format || b.format === 'auto' ? 'Choisis le format le plus adapté au sujet.' : `Format imposé : « ${b.format} ».`);
  const p = b.previous;
  if (p && p.title) ctx += `\nCe cours est la SUITE d'un cours déjà écouté intitulé « ${String(p.title).slice(0, 120)} » dont voici le résumé : ${String(p.summary || '').slice(0, 600)}. Ne répète pas ce qui a été vu, commence par un très court rappel puis va plus loin.`;
  return ctx;
};
const sizeOf = (minutes) => {
  const min = Math.min(25, Math.max(3, Number(minutes) || 10));
  const nSec = Math.max(2, Math.min(7, Math.round(min / 2.5)));
  return { min, nSec, wordsPer: Math.round((min * 150) / nSec) };
};
const STAGE = /^\W*(pause|silence|musique|jingle|rire|\(|\[)/i;

async function planOf(b, env) {
  if (!b.topic || typeof b.topic !== 'string') throw Object.assign(new Error('Sujet manquant'), { status: 400 });
  const { min, nSec, wordsPer } = sizeOf(b.minutes);
  const llm = makeLLM(env);
  const outline = await llm(
    `${ctxOf(b)}\nDurée totale visée : ${min} minutes. Prépare le PLAN du cours en exactement ${nSec} parties (le récap, le quiz et la conclusion seront ajoutés à la dernière partie). Pour chaque partie : un titre court (heading) et 3 à 4 idées clés à couvrir (points), en les ordonnant du plus simple au plus avancé. Le champ « summary » résume en 2 phrases ce que le cours couvre. Le champ « next » propose 3 idées de suite courtes. « emoji » : un seul emoji représentatif. Choisis aussi le format.`,
    OUTLINE_SCHEMA,
    (o) => o && Array.isArray(o.sections) && o.sections.length && o.title
  );
  const fmt = ['prof', 'duo', 'story'].includes(outline.format) ? outline.format : (['prof', 'duo', 'story'].includes(b.format) ? b.format : 'prof');
  return {
    title: String(outline.title).slice(0, 140), emoji: String(outline.emoji || '🎧').slice(0, 8), format: fmt,
    summary: String(outline.summary || ''), next: (Array.isArray(outline.next) ? outline.next : []).slice(0, 3).map(String),
    sections: outline.sections.slice(0, nSec).map((s) => ({ heading: String(s.heading || ''), points: String(s.points || '') })), wordsPer,
  };
}

async function sectionOf(b, env) {
  const secs = Array.isArray(b.sections) ? b.sections : [];
  const i = Number(b.i) || 0, s = secs[i];
  if (!s) throw Object.assign(new Error('Partie inconnue'), { status: 400 });
  const wordsPer = Math.min(700, Math.max(80, Number(b.wordsPer) || 250));
  const fmt = b.outlineFormat || 'prof';
  const plan = secs.map((x, j) => `${j + 1}. ${x.heading} (${x.points})`).join(' | ');
  const first = i === 0, last = i === secs.length - 1;
  const llm = makeLLM(env);
  const r = await llm(
    `${ctxOf(b)}\nFormat du cours : « ${fmt} » (en « duo » alterne les voix A et B ; sinon uniquement la voix A).\nPlan complet du cours : ${plan}\nTu écris UNIQUEMENT la partie ${i + 1}/${secs.length} : « ${s.heading} », qui couvre : ${s.points}.\nLongueur : environ ${wordsPer} mots, jamais moins de ${Math.round(wordsPer * 0.8)} mots.\n` +
      (first ? 'Commence par une accroche forte dans les 15 premières secondes.' : 'Ne dis pas bonjour : enchaîne naturellement, comme si la partie précédente venait de se terminer, sans la résumer.') + '\n' +
      (last
        ? "Termine par un récapitulatif de l'essentiel, puis un mini quiz oral de 3 questions : chaque question est un segment à part avec « pause » de 6 secondes, suivie de la réponse en un segment court. Le TOUT DERNIER segment est une CONCLUSION de 2 à 3 phrases qui résume tout le podcast et le conclut chaleureusement."
        : "Ne conclus pas et ne fais pas de récap : une autre partie suit.") +
      "\nN'écris jamais d'indications de mise en scène (« pause de six secondes », « silence », « musique ») dans le texte parlé. Chaque segment fait 1 à 3 paragraphes parlés (max ~700 caractères).",
    SEG_SCHEMA,
    (o) => o && Array.isArray(o.segments) && o.segments.some((x) => x && typeof x.text === 'string' && x.text.trim().length > 20)
  );
  const segments = r.segments
    .filter((x) => x && typeof x.text === 'string' && x.text.trim() && !STAGE.test(x.text.trim()))
    .map((x) => {
      const o = { speaker: fmt === 'duo' && x.speaker === 'B' ? 'B' : 'A', text: x.text.trim() };
      const p = Number(x.pause); if (p > 0) o.pause = Math.min(12, p);
      return o;
    });
  if (!segments.length) throw new Error('Partie vide');
  return segments;
}

const fail = (e) => json({ error: String((e && e.message) || e) }, (e && e.status) || 502);
async function handlePlan(req, env) { try { return json(await planOf(await req.json(), env)); } catch (e) { return fail(e); } }
async function handleSection(req, env) { try { return json({ segments: await sectionOf(await req.json(), env) }); } catch (e) { return fail(e); } }
async function handleScript(req, env) {
  try {
    const b = await req.json(); const plan = await planOf(b, env);
    const parts = await Promise.all(plan.sections.map((_, i) => sectionOf({ ...b, outlineFormat: plan.format, sections: plan.sections, i, wordsPer: plan.wordsPer }, env)));
    return json({ title: plan.title, emoji: plan.emoji, format: plan.format, summary: plan.summary, next: plan.next, segments: parts.flat() });
  } catch (e) { return fail(e); }
}

/* ------------------------------------------------------------------ Voix */

const TTS_MODELS = ['gemini-3.8-flash-tts', 'gemini-3.8-flash-lite-tts', 'gemini-3.1-flash-tts-preview', 'gemini-2.5-flash-preview-tts', 'gemini-2.5-pro-preview-tts'];
const TTS_DEAD = new Map();
const VOICES = new Set(['Puck', 'Charon', 'Fenrir', 'Orus', 'Kore', 'Aoede', 'Leda', 'Zephyr']);

function pcmToWav(pcm, rate) {
  const buf = new Uint8Array(44 + pcm.length), v = new DataView(buf.buffer);
  const w = (o, s) => { for (let i = 0; i < s.length; i++) v.setUint8(o + i, s.charCodeAt(i)); };
  w(0, 'RIFF'); v.setUint32(4, 36 + pcm.length, true); w(8, 'WAVE'); w(12, 'fmt '); v.setUint32(16, 16, true);
  v.setUint16(20, 1, true); v.setUint16(22, 1, true); v.setUint32(24, rate, true); v.setUint32(28, rate * 2, true);
  v.setUint16(32, 2, true); v.setUint16(34, 16, true); w(36, 'data'); v.setUint32(40, pcm.length, true);
  buf.set(pcm, 44);
  return buf;
}

async function handleTTS(req, env) {
  const { turns, text, voices } = await req.json();
  const list = (Array.isArray(turns) && turns.length ? turns : [{ s: 'A', t: String(text || '') }]).map((x) => ({ s: x.s === 'B' ? 'B' : 'A', t: String(x.t || '').trim() })).filter((x) => x.t);
  const total = list.reduce((n, x) => n + x.t.length, 0);
  if (!total || total > 3500) return json({ error: 'Texte invalide ou trop long' }, 400);
  const pick = (v, d) => (VOICES.has(v) ? v : d);
  const A = pick(voices && voices.A, 'Puck'), B = pick(voices && voices.B, 'Kore');
  const multi = new Set(list.map((x) => x.s)).size > 1;
  const body = multi ? list.map((x) => x.s + ': ' + x.t).join('\n') : list.map((x) => x.t).join(' ');
  const vc = (n) => ({ prebuiltVoiceConfig: { voiceName: n } });
  const speechConfig = multi
    ? { multiSpeakerVoiceConfig: { speakerVoiceConfigs: [{ speaker: 'A', voiceConfig: vc(A) }, { speaker: 'B', voiceConfig: vc(B) }] } }
    : { voiceConfig: vc(list[0].s === 'B' ? B : A) };
  const minBytes = Math.max(4000, total * 22 * 48); // au moins ~ total/22 secondes d'audio (24 kHz, 16 bits)
  const t0 = Date.now();
  const errs = [];
  while (Date.now() - t0 < 30000) {
    for (const m of TTS_MODELS) {
      if ((TTS_DEAD.get(m) || 0) > Date.now()) continue;
      try {
        const r = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${m}:generateContent`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', 'x-goog-api-key': env.GEMINI_API_KEY },
          signal: AbortSignal.timeout(75000),
          body: JSON.stringify({ contents: [{ parts: [{ text: body }] }], generationConfig: { responseModalities: ['AUDIO'], speechConfig } }),
        });
        if (r.status === 429) {
          const t = await r.text(); const rd = /"retryDelay":\s*"(\d+)/.exec(t);
          const daily = /PerDay/i.test(t) && !/PerMinute/i.test(t);
          TTS_DEAD.set(m, Date.now() + (daily ? 3600000 : Math.min(60, Math.max(8, rd ? +rd[1] : 20)) * 1000)); errs.push(m + ' quota'); continue;
        }
        if (!r.ok) { errs.push(m + ' ' + r.status); TTS_DEAD.set(m, Date.now() + 5000); continue; }
        const d = await r.json();
        const parts = (d && d.candidates && d.candidates[0] && d.candidates[0].content && d.candidates[0].content.parts) || [];
        const inl = (parts.find((p) => p.inlineData && p.inlineData.data) || {}).inlineData;
        if (!inl) { errs.push(m + ' vide'); continue; }
        const bin = atob(inl.data), pcm = new Uint8Array(bin.length);
        for (let i = 0; i < bin.length; i++) pcm[i] = bin.charCodeAt(i);
        if (pcm.length < minBytes) { errs.push(m + ' audio trop court'); continue; }
        return new Response(pcmToWav(pcm, 24000), { headers: { 'Content-Type': 'audio/wav', 'X-Engine': m, ...cors } });
      } catch (e) { errs.push(m + ' ' + String((e && e.message) || e).slice(0, 60)); }
    }
    // tous les modèles sont en pause (quota par minute) : on attend le premier qui revient
    const next = Math.min(...TTS_MODELS.map((m) => TTS_DEAD.get(m) || 0));
    const wait = Math.min(15000, Math.max(1500, next - Date.now()));
    if (Date.now() - t0 + wait > 30000) break;
    await sleep(wait);
  }
  try {
    const mr = await env.AI.run('@cf/myshell-ai/melotts', { prompt: list.map((x) => x.t).join(' '), lang: 'fr' });
    const b64 = typeof mr === 'string' ? mr : mr && mr.audio;
    if (b64) {
      const bin = atob(b64), u8 = new Uint8Array(bin.length);
      for (let i = 0; i < bin.length; i++) u8[i] = bin.charCodeAt(i);
      if (u8.length > 2000) return new Response(u8, { headers: { 'Content-Type': 'audio/mpeg', 'X-Engine': 'melotts', ...cors } });
    }
    errs.push('melotts vide');
  } catch (e) { errs.push('melotts ' + String((e && e.message) || e).slice(0, 80)); }
  return json({ error: 'Voix momentanément saturée : ' + errs.slice(-4).join(' ; ') }, 503);
}
